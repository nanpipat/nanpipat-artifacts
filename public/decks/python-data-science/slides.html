import pandas as pd
import numpy as np
np.random.seed(42)
n = 200
menus = ["Latte", "Mocha", "Espresso", "Matcha",
         "Americano", "Thai Tea", "Cocoa"]
# prices · types · days → dict ราคา / ร้อน-เย็น / จันทร์–อาทิตย์
menus_list = np.random.choice(menus, n)
data = {"วันที่": …, "เมนู": menus_list, "ประเภท": …, "ราคา": …,
        "จำนวน": np.random.randint(1, 8, n).astype(float)}
df = pd.DataFrame(data)
# ใส่ข้อมูลหาย (NaN) สัก 5% เพื่อฝึก Data Cleaning ทีหลัง
mask = np.random.random(n) < 0.05
df.loc[mask, "จำนวน"] = np.nan
df.to_csv("coffee_shop.csv", index=False)
print("✅ สร้าง dataset เสร็จแล้ว!", df.shape)

import pandas as pd
 
# สร้างข้อมูลในรูปแบบ Dictionary (คีย์คือชื่อคอลัมน์)

ฟอร์มที่ลูกค้ากรอกเอง: บางคนลืมกรอก บางคนกรอกเลขเป็น text บางวันบันทึกซ้ำ

0–0%

ของเวลาทำงาน Data Scientist หมดไปกับการ clean ข้อมูล

❗ ข้อมูลหาย (NaN)

ลืมกรอกจำนวน เช็คdf.isnull().sum()แก้fillna() dropna()

❗ Data type ผิด

เช่น ราคาเป็น text เช็คdf["ราคา"].dtypeแก้df["ราคา"].astype(float)

❗ ข้อมูลซ้ำ

duplicate · บันทึกซ้ำ เช็คdf.duplicated().sum()แก้df.drop_duplicates()

import numpy as np
dirty = pd.DataFrame({
    "ชื่อ": ["ไอซ์", "มิ้น", None, "แพร", "บีม"],
    "คะแนน": [85, None, 78, 95, None],
    "สาขา": ["CS", "IT", "CS", None, "DS"]
})
print(dirty)
print("---")
print(dirty.isnull().sum())
# เติมค่าเฉลี่ยแทน NaN
avg = dirty["คะแนน"].mean()
dirty["คะแนน"] = dirty["คะแนน"].fillna(avg)
print(dirty)

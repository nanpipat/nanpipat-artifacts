total_fuel  = distance_km * fuel_per_km
total_cost  = total_fuel * fuel_price
cost_each   = total_cost / num_people
 
print("=== คำนวณค่าเดินทาง ===")
print(f"ระยะทาง: {distance_km} กม.")
print(f"น้ำมันที่ใช้: {total_fuel:.2f} ลิตร")
print(f"ค่าน้ำมันรวม: {total_cost:.2f} บาท")
print(f"หารกัน {num_people} คน คนละ: {cost_each:.2f} บาท")

ปกติ Python รันจากบนลงล่างทีละบรรทัด — Control Structure ช่วยคุมว่าจะทำอะไร เมื่อไร และกี่รอบ

🛒“ถ้าเงินพอก็ซื้อ ถ้าไม่พอก็ไม่ซื้อ” = ตัดสินใจ · “ส่งอีเมลให้ลูกค้าทุกคน” = ทำซ้ำ

if / elif / else

if เงื่อนไข:
    # ทำเมื่อ True
elif เงื่อนไขอื่น:
    # ทำเมื่อข้อแรก False
else:
    # ทำเมื่อไม่ตรงเลย
ifถ้าเงื่อนไขเป็น True ก็ทำ
elifถ้า if ข้อแรกเป็น False แต่ข้อนี้ True ก็ทำ
elseถ้าไม่ตรงเงื่อนไขไหนเลย ก็ทำอันนี้
⚠️หลังเงื่อนไขต้องมี : เสมอ และโค้ดข้างในต้องเยื้องเข้า 4 spaces
